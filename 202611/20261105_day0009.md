### class
# 클래스 정의(선언) : 붕어빵 틀
class User:
    # __init__(생성자) : 클래스가 처음 생성될 때 자동으로 호출된다.
    def __init__ (self, name) :
        self.name = name # 변수를 초기화

    def show(self): # 메서드(Method)
        return f'회원 이름 : {self.name}'

# 인스턴스 생성 : 붕어빵 만들기
user_a = User('최진웅') # init에서 self는 선언했기 때문에 파라미터에 {name}만 들어가도 된다.
user_b = User('김태호')

# User 클래스로 생성한 인스턴스(메모리에  생성)

# 클래스의 인스턴스 변수
print(user_a.name)
print(user_b.name)


# 클래스 메서드 접근
print(user_a.show()) # 클래스 안에서 생성한 함수를 불러오려면 `instance`.`def_name`처럼 .을 이용하면 된다.
print(user_b.show()) 

print(type(user_a))
print(user_a is user_b)
print(id(user_a))
print(id(user_b))

class Monitor:
    def display_info(self):
        # 인스턴스 변수
        self.resolution = '1920X1080'
        self.size = '27inch'
        print(f'모니터 정보\n해상도: {self.resolution}\n크기: {self.size}\n객체 ID: {id(self)}')

monitor = Monitor() # 객체(인스턴스) 생성(모니터 객체 생성)
print(id(monitor))  # 모니터 인스턴스의 ID 출력
